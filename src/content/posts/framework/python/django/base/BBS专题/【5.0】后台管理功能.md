# 【一】前提
+ 我们在书写模型表时，已经将所有的模型表注册到了对应 APP 下的 admin.py文件中

```python
class BaseModelAdmin(admin.ModelAdmin):
  # display_字段名(self,obj)
    def display_created_time(self, obj):
      # 将需要格式化后的样式返回
        return obj.create_time.strftime("%Y-%m-%d %H:%M:%S")
		# 定制后台字段展示文字和 verbose_name 同理
    display_created_time.short_description = "创建时间"

    def display_update_time(self, obj):
        return obj.update_time.strftime("%Y-%m-%d %H:%M:%S")

    display_update_time.short_description = "更新时间"
```

## 【2】模型表示例
```python
from django.contrib import admin
from django.utils.html import format_html

from lib.CommonAdmin import BaseModelAdmin
from user.models import UserInfo


# Register your models here.

# 创建一个类并用它来定制UserInfo

class UserInfoAdmin(BaseModelAdmin):
    def display_avatar(self, obj):

        avatar_path = obj.avatar if obj.avatar else 'static/advImg/default.png'
        # 拼接完整的URL
        avatar_url = f'http://127.0.0.1:8000/{avatar_path}'

        return format_html(
            f'<a href="{avatar_url}" target="_blank">'
            f'<img src="{avatar_url}" alt="请稍后再试哦" style="width: 50px;height: auto">'
            f'</a>')

    display_avatar.short_description = "头像"  # 设置列名显示为"Avatar"

                    'is_deleted',
```

## 【2】注册个人站点模型表后台
+ blog/admin.py

```python
from django.contrib import admin
from django.utils.html import format_html
from lib.CommonAdmin import BaseModelAdmin

# Register your models here.
from blog.models import Blog, Adv


@admin.register(Blog)
class BlogAdmin(BaseModelAdmin):
    ordering = ["id"]

    list_display = ['id', 'site_name', 'site_title', 'display_created_time', 'display_update_time', 'site_theme']


@admin.register(Adv)
class AdvAdmin(BaseModelAdmin):
    ordering = ["id"]

    def display_img(self, obj):
        # 获取默认的avatar路径
        img_path = obj.img if obj.img else 'static/advImg/default.png'
        # 拼接完整的URL
        img_url = f'http://127.0.0.1:8000/media/{img_path}'
        # 格式化为HTML链接，并返回
        # return format_html('<img href=>{}</a>', avatar_url, img_url)

        return format_html(
            f'<a href="{img_url}" target="_blank">'
            f'<img src="{img_url}" alt="请稍后再试哦" style="width: 50px;height: auto">'
            f'</a>')
		# 设置列名显示为"广告图片"
    display_img.short_description = "广告图片"  
    list_display = ['id', 'title', 'content', 'display_created_time', 'display_update_time', 'mobile',
                    'is_background_img', 'display_img']
```

## 【3】注册文章模型表后台
+ article/admin.py

```python
from django.contrib import admin

# Register your models here.
from article.models import CategoryClass, CategoryTag, Comment, UpAndDown, Article, ArticleToTag
from lib.CommonAdmin import BaseModelAdmin


@admin.register(Article)
class ArticleAdmin(BaseModelAdmin):
    ordering = ["id"]

    def display_content(self, obj):
        return obj.content[:30]

    display_content.short_description = "文章内容"
    list_display = ['id', 'title', 'desc', 'up_num', 'down_num', 'comment_num', 'display_content',
                    'display_created_time', 'display_update_time', 'blog', 'category']


@admin.register(CategoryClass)
class CategoryClassAdmin(BaseModelAdmin):
    ordering = ["id"]
    list_display = ['id', 'name', 'blog']


@admin.register(CategoryTag)
class CategoryTagAdmin(BaseModelAdmin):
    ordering = ["id"]
    list_display = ['id', 'name', 'blog']


@admin.register(ArticleToTag)
class CategoryTagAdmin(admin.ModelAdmin):
    ordering = ["id"]
    list_display = ['id', 'article', 'tag']


@admin.register(UpAndDown)
class UpAndDownAdmin(BaseModelAdmin):
    ordering = ["id"]
    list_display = ['id', 'user', 'article', 'up_or_down', 'display_created_time', 'display_update_time']


@admin.register(Comment)
class CommentAdmin(BaseModelAdmin):
    ordering = ["id"]
    list_display = ['id', 'user', 'article', 'content', 'is_deleted', 'parent', 'display_created_time',
                    'display_update_time']
```
